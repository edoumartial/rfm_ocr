-- ==========================================
-- 1. TABLES DE BASE ET GESTION DES UTILISATEURS
-- ==========================================

CREATE TABLE IF NOT EXISTS public.users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(100) NOT NULL,
    password_hash VARCHAR(100) NOT NULL,
    role VARCHAR(50),
    is_active BOOLEAN DEFAULT TRUE
);

-- ==========================================
-- 2. GESTION GÉOGRAPHIQUE (CADASTRE)
-- ==========================================

CREATE TABLE IF NOT EXISTS public.cadastre_province (
    id BIGSERIAL PRIMARY KEY,
    nom VARCHAR(100) NOT NULL
);

CREATE TABLE IF NOT EXISTS public.cadastre_commune (
    id BIGSERIAL PRIMARY KEY,
    nom VARCHAR(100) NOT NULL,
    province_id BIGINT REFERENCES public.cadastre_province(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS public.cadastre_dossierfoncier (
    id BIGSERIAL PRIMARY KEY,
    ref_dossier VARCHAR(50),
    nom_titulaire VARCHAR(255),
    section VARCHAR(50),
    parcelle VARCHAR(50),
    statut_brut TEXT,
    document_pdf VARCHAR(100),
    date_creation TIMESTAMP WITH TIME ZONE,
    commune_id BIGINT REFERENCES public.cadastre_commune(id) ON DELETE SET NULL,
    statut VARCHAR(20)
);

CREATE TABLE IF NOT EXISTS public.cadastre_journalmodifications (
    id BIGSERIAL PRIMARY KEY,
    champ_modifier VARCHAR(30),
    ancienne_valeur TEXT,
    nouvelle_valeur TEXT,
    type_action VARCHAR(20),
    date_modification TIMESTAMP WITH TIME ZONE,
    auteur VARCHAR(150),
    dossier_id BIGINT REFERENCES public.cadastre_dossierfoncier(id) ON DELETE CASCADE
);

-- ==========================================
-- 3. GESTION DES DOCUMENTS ET DONNÉES OCR (RFM)
-- ==========================================

CREATE TABLE IF NOT EXISTS public.rfm_documents (
    id SERIAL PRIMARY KEY,
    user_id INTEGER,
    filename VARCHAR(255),
    uploaded_by VARCHAR(100),
    created_at TIMESTAMP WITHOUT TIME ZONE,
    extraction_ocr TEXT,
    statut_caracter VARCHAR(50),
    valide_par VARCHAR(100),
    validated_at TIMESTAMP WITHOUT TIME ZONE,
    upload_date TIMESTAMP WITHOUT TIME ZONE
);

CREATE TABLE IF NOT EXISTS public.rfm_data (
    id SERIAL PRIMARY KEY,
    document_id INTEGER REFERENCES public.rfm_documents(id) ON DELETE CASCADE,
    lieudict VARCHAR(255),
    province VARCHAR(100),
    commune VARCHAR(100),
    parcelle VARCHAR(100),
    section VARCHAR(100),
    requerant VARCHAR(255),
    reference VARCHAR(100)
);

-- ==========================================
-- 4. TABLES AUTHENTIFICATION ET SYSTEME (DJANGO)
-- ==========================================

CREATE TABLE IF NOT EXISTS public.django_content_type (
    id SERIAL PRIMARY KEY,
    app_label VARCHAR(100) NOT NULL,
    model VARCHAR(100) NOT NULL
);

CREATE TABLE IF NOT EXISTS public.auth_permission (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    content_type_id INTEGER REFERENCES public.django_content_type(id) ON DELETE CASCADE,
    codename VARCHAR(100) NOT NULL
);

CREATE TABLE IF NOT EXISTS public.auth_group (
    id SERIAL PRIMARY KEY,
    name VARCHAR(150) NOT NULL UNIQUE
);

CREATE TABLE IF NOT EXISTS public.auth_group_permissions (
    id BIGSERIAL PRIMARY KEY,
    group_id INTEGER REFERENCES public.auth_group(id) ON DELETE CASCADE,
    permission_id INTEGER REFERENCES public.auth_permission(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS public.auth_user (
    id SERIAL PRIMARY KEY,
    password VARCHAR(128) NOT NULL,
    last_login TIMESTAMP WITH TIME ZONE,
    is_superuser BOOLEAN NOT NULL,
    username VARCHAR(150) NOT NULL UNIQUE,
    first_name VARCHAR(150) NOT NULL,
    last_name VARCHAR(150) NOT NULL,
    email VARCHAR(254) NOT NULL,
    is_staff BOOLEAN NOT NULL,
    is_active BOOLEAN NOT NULL,
    date_joined TIMESTAMP WITH TIME ZONE NOT NULL
);

CREATE TABLE IF NOT EXISTS public.auth_user_groups (
    id BIGSERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES public.auth_user(id) ON DELETE CASCADE,
    group_id INTEGER REFERENCES public.auth_group(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS public.auth_user_permissions (
    id BIGSERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES public.auth_user(id) ON DELETE CASCADE,
    permission_id INTEGER REFERENCES public.auth_permission(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS public.django_migrations (
    id SERIAL PRIMARY KEY,
    app VARCHAR(255) NOT NULL,
    name VARCHAR(255) NOT NULL,
    applied TIMESTAMP WITH TIME ZONE NOT NULL
);

CREATE TABLE IF NOT EXISTS public.django_session (
    session_key VARCHAR(40) PRIMARY KEY,
    session_data TEXT NOT NULL,
    expire_date TIMESTAMP WITH TIME ZONE NOT NULL
);

CREATE TABLE IF NOT EXISTS public.django_admin_log (
    id SERIAL PRIMARY KEY,
    action_time TIMESTAMP WITH TIME ZONE NOT NULL,
    object_id TEXT,
    object_repr VARCHAR(200) NOT NULL,
    action_flag SMALLINT NOT NULL,
    change_message TEXT NOT NULL,
    content_type_id INTEGER REFERENCES public.django_content_type(id) ON DELETE SET NULL,
    user_id INTEGER REFERENCES public.auth_user(id) ON DELETE CASCADE
);
